\begin{definition}{Podstawowe Oznaczenia}{}{}
    \begin{itemize}
        \item $x = x(t)$ - wektor stanu
        \item $u = u(t)$ - wektor sterujący
        \item $z = z(t)$ - wektor zakłóceń
    \end{itemize}
\end{definition}

\begin{definition}{Model Matematyczny}{}{}
    Model charakteryzujący się opisywaniem procesu / dynamiki.
    \[
    \begin{cases}
        \dot{x}(t) = f(t, x, u, z) \\
        x(t_0) = x_0
    \end{cases}
    \]
\end{definition}

\begin{definition}{Funkcja celu / zysku / wygody (cel sterowania)}{}{}
    $$J(u) = \int_{t_0}^{t_k} F(x(t), u(t), z(t)) dt$$
\end{definition}

\begin{example}{Zagadnienie Sterowania Optymalnego w ekonomii}{}{}
    Niech $x(t)$ określa zysk pewnej firmy w czasie $t$. Niech $u = u(t)$, gdzie $0 \leq u(t) \leq 1$ określa część zysku przeznaczoną na produkcję ($u = 0$ - całość przeznaczona na konsumpcję, $u = 1$ - całość przeznaczona na produkcję). \\
    Naszym modelem matematycznym (przy danym współczynniku $k$) będzie: $$\dot{x} = k u x$$
    Zadanie polega na znalezieniu $u = u(t)$ takiego, aby całkowita konsumpcja na pewnym odcinku czasowym $[0, t_1]$, czyli $$J(u) = \int^{t_1}_0 (1 - u(t)) x(t) dt$$
    była jak największa.
\end{example}

\begin{example}{Zagadnienie czaso-optymalnego sterowania}{}{}
    Niech: \begin{itemize}
        \item $x$ oznacza kąt obrotu wału silnika, 
        \item $t$ oznacza czas
        \item $u$ oznacza napięcie przyłożone do zacisków silnika
    \end{itemize}
    Należy przeprowadzić wał silnika z położenia $x_0$ do położenia $x_1$ w najkrótszym czasie, przy czym napięcie $|u| \leq a$.
    Nasz proces definiujemy następująco:
    $$
    \begin{cases}
        \ddot{x}(t) = u(t) \\
        \dot{x}(0) = 0 \\
        \dot{x}(t_k) = 0 \\
        x(0) = x_0 \\
        x(t_k) = x_1 \\
        |u(t)| \leq 0 \\
    \end{cases}
    $$
    Nasze zadanie polega na wyznaczeniu takiej funkcji $u(t)$ aby $t_k$ było najmniejsze.

    \begin{solution}{Feldbaum - Rozwiązanie typu bang-bang}{}{}
        $$u(t) = \begin{cases}
            a & \text{for } 0 \leq t \leq \frac{t_k}{2} \\
            -a & \text{for } \frac{t_k}{2} \leq t \leq t_k \\
        \end{cases}$$
    \end{solution}
\end{example}

\begin{example}{Sterowanie optymalne w biologii (wzrost rośliny)}{}{}
    Niech \begin{itemize}
        \item $x = x(t)$ - wysokość rośliny w chwili $t$
        \item $u = u(t)$ - dodatkowy przyrost rośliny spowodowany sztucznym oświetleniem rośliny
        \item $\dot{x} = 1 + u$ - model
        \item $x(0) = 0$
        \item $x(1) = 2$
        \item $J(u) = \int_0^1 \frac{u^2(t)}{2} dt$ - Koszt sztucznego oświetlenia
    \end{itemize}
    Szukamy takiego sterowania $u$, że odpowiednie rozwiązanie spełniające warunki brzegowe $J$ przyjmuje najmniejszą wartość.

    \begin{solution}{}{}{}
        $$x(t) - x(0) = \int^t_0 (1 + u(s)) ds$$
        Zauważmy, że $\int^1_0 x(t) dt = 1$, a także:
        \[\int^1_0 \frac{(u(t) - 1)^2}{2} dt = \int^1_0 (\frac{u^2(t)}{2} - \frac{2 u(t)}{2} + \frac{1}{2}) dt = \int^1_0 \frac{u^2(t)}{2} dt - \int^1_0 u(t) dt + \frac{1}{2} = J(u) - \frac{1}{2}\]
        A więc zachodzi \[J(u) = \int_0^1 \frac{u^2(t)}{2} dt = \int^1_0 \frac{(u(t) - 1)^2}{2} dt + \frac{1}{2}\]
    \end{solution}

    Minimum - $J(u) = \frac{1}{2}$ (osiągnięte dla $u \equiv 0$ na [0, 1]) \\
    Optymalne rozwiązanie - $x(t) = \int^1_0 (1 + 1) dt = 2t$, gdzie $t \in [0, 1]$
\end{example}
